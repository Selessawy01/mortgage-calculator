import { useState }          from 'react'
import { calculateMortgage } from './utils/calculateMortgage'
import { MORTGAGE_TYPES }    from './utils/constants'




function MortgageForm({ onCalculate, onClear}) {

  const [mortgageAmount, setMortgageAmount] = useState('')
  const [mortgageTerm, setMortgageTerm] = useState('')
  const [mortgageInterest, setMortgageInterest] = useState('')
  const [mortgageType, setMortgageType] = useState('')
  const [errors, setErrors] = useState({})
  

  function handleClear() {
    setMortgageAmount('')
    setMortgageTerm('')
    setMortgageInterest('')
    setMortgageType('')
    setErrors({})

    onClear()

  }

  function handleChange(event, setter, fieldName) {
  const value = event.target.value

  setter(value)

  if (errors[fieldName]) {
    setErrors((prevErrors) => ({
      ...prevErrors,
      [fieldName]: ''
    }))
  }
}

  function validateForm(){
    const newErrors = {}

    if (mortgageAmount === ''){
         newErrors.mortgageAmount = 'This field is required'
    } else if (Number(mortgageAmount) <= 0) {
         newErrors.mortgageAmount = 'Enter an amount greater than 0'}

    if (mortgageTerm === ''){
         newErrors.mortgageTerm = 'This field is required'
    } else if (Number(mortgageTerm) <= 0) {
        newErrors.mortgageTerm = 'Enter an amount greater than 0'
    }

    if (mortgageInterest === '') {
       newErrors.mortgageInterest = 'This field is required'
    } else if (Number(mortgageInterest) < 0) {
       newErrors.mortgageInterest = 'Enter a rate of 0 or more'
    }
    if (!mortgageType) {
    newErrors.mortgageType = 'This field is required'
    }

    setErrors(newErrors)
    
    return Object.keys(newErrors).length === 0
  }

  function handleSubmit(event){
    event.preventDefault();

    if (!validateForm()) return

    const calculatedResults = calculateMortgage({
      amount       : mortgageAmount, 
      term         : mortgageTerm, 
      interest     : mortgageInterest,
      type         : mortgageType})
    
    onCalculate(calculatedResults)
  }

  return (
    <section className="calculator">
      <form className="form"  onSubmit={handleSubmit} noValidate>
        <div className="form__header">
          <h1 className="form__title xl-text">Mortgage Calculator</h1>
          <button type="button" className="form__clear md-text" onClick={handleClear}>Clear All</button>
        </div>

        <div className="form__fields">
         <div className="form__group">
          <label className="form__label md-text" htmlFor="mortgage-amount">Mortgage Amount</label>
          <div className={`form__input-wrapper ${errors.mortgageAmount ? 'error' : ''}`}>
            <span className={`form__prefix lg-text ${errors.mortgageAmount ? 'error' : ''}`}>£</span>
            <input className='form__input lg-text' type="number" id="mortgage-amount" name ="mortgage-amount" min="1" required   value={mortgageAmount} 
                   onChange={(event) => handleChange(event, setMortgageAmount, 'mortgageAmount')}   
                   placeholder=" "
                   aria-invalid={Boolean(errors.mortgageAmount)}  aria-describedby={errors.mortgageAmount ? "amount-error" : undefined}/>
          </div>
         {errors.mortgageAmount &&  ( <p id="amount-error" className="form__error sm-text" role="alert"> {errors.mortgageAmount}</p> )}

         </div>

        <div className="form__group-values">
         <div className="form__group">
          <label className="form__label md-text" htmlFor="mortgage-term">Mortgage Term</label>
          <div  className={`form__input-wrapper ${errors.mortgageTerm ? 'error' : ''}`}>
             <input className="form__input lg-text" type="number" id="mortgage-term" name ="mortgage-term" min="1" value={mortgageTerm} 
                    onChange={(event) => handleChange(event, setMortgageTerm, 'mortgageTerm')} 
                    placeholder=" "
                    aria-invalid={Boolean(errors.mortgageTerm)} aria-describedby={errors.mortgageTerm ? "term-error" : undefined} />
             <span className={`form__suffix lg-text ${errors.mortgageTerm ? 'error' : ''} `}>years</span>
          </div>
         {errors.mortgageTerm && (<p id="term-error" className="form__error sm-text" role="alert">{errors.mortgageTerm}</p> )}
         </div>
         <div className="form__group">
          <label className="form__label md-text" htmlFor="interest-rate">Interest Rate</label>
          <div className={`form__input-wrapper ${errors.mortgageInterest ? 'error' : ''}`}>
            <input className="form__input lg-text" type="number" id="interest-rate" name ="interest-rate" min="0" step="any" required
                   value={mortgageInterest}  placeholder =" "
                   onChange={(event) => handleChange(event, setMortgageInterest, 'mortgageInterest')}
                   aria-invalid={Boolean(errors.mortgageInterest)} aria-describedby={errors.mortgageInterest ? "interest-error" : undefined}/>
            <span className={`form__suffix lg-text ${errors.mortgageInterest ? 'error' : ''} `}>%</span>
          </div>
          { errors.mortgageInterest  && ( <p id="interest-error"className="form__error sm-text" role="alert">{errors.mortgageInterest}</p> )}
         </div>
        </div> 
         <fieldset className="form__group form__group--type"
                   aria-describedby={errors.mortgageType ? "type-error" : undefined}>
            <legend className="form__label md-text">Mortgage Type</legend>
            <div className="form__options">
               <label className="form__option lg-text" htmlFor="repayment">
                <input type="radio" id="repayment" name="mortgage_type" value={MORTGAGE_TYPES.REPAYMENT} required
                       checked={mortgageType === MORTGAGE_TYPES.REPAYMENT}
                       onChange={(event) =>handleChange(event, setMortgageType, 'mortgageType')}/>
                  Repayment
               </label>
               <label className="form__option lg-text" htmlFor="interest">
                <input type="radio" id="interest" name="mortgage_type" value={ MORTGAGE_TYPES.INTEREST_ONLY} required
                        checked={mortgageType === MORTGAGE_TYPES.INTEREST_ONLY}
                        onChange={(event) =>handleChange(event, setMortgageType, 'mortgageType')} />
                 Interest Only
               </label>
            </div>
            { errors.mortgageType && ( <p id="type-error" className="form__error sm-text" role="alert">{errors.mortgageType}</p> )}
         </fieldset>
        </div>

        <button type="submit" className="form__submit lg-text">
          <img src="./assets/images/icon-calculator.svg" alt=""/> Calculate Repayments
        </button>
      </form>

    </section>
  )
}

export default MortgageForm




function MortgageResults({ results }) {
  return (
          <section className="results" aria-live="polite" aria-atomic="true">
      { !results ? ( 
       <div className="results__empty">
         <img className="results__image" src="./assets/images/illustration-empty.svg" alt=""/>
         <h2 className="results__title xl-text">Results shown here</h2>
         <p  className="results__paragraph md-text">Complete the form and click “calculate repayments” to see what your monthly repayments would be.</p>
       </div>
      ) : (
       <div className="results__output">
        <div className="results__header">
          <h2 className="results__title  xl-text">Your results</h2>
          <p  className="results__paragraph md-text">Your results are shown below based on the information you provided. To adjust the results, edit the form and click “calculate repayments” again.</p>
        </div>
        <div className="results__result">
          <div className="results__monthly">
            <p className="results__monthly-title md-text">Your monthly repayments</p>
            <p className="results__monthly-amount amount-text ">£{results.monthlyPayment.toLocaleString('en-GB', {minimumFractionDigits: 2,maximumFractionDigits: 2})}</p>
          </div>
          <div className="results__separator"></div>
          <div className="results__total">
            <p className="results__total-title md-text">Total you'll repay over the term</p>
            <p className="results__total-amount xl-text">£{results.totalRepayment.toLocaleString('en-GB', {minimumFractionDigits: 2,maximumFractionDigits: 2})}</p>
          </div>
        </div>
       </div>
      )}
    </section>

  )}
  export default MortgageResults
